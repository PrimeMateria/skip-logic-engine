\title{Skip logic domain compilation pipeline and topological verifier}
\begin{semanticmodule}
structure{
  name: CompilationPipelineStatus,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: totalTiersCount, type: Nat},
    {name: deadBranchesPruned, type: Nat},
    {name: simdVectorWidthBits, type: Nat},
    {name: zeroCostRustEmitted, type: Bool}
  ]
}

definition{
  name: defaultPipelineStatus,
  parameters: [],
  result: CompilationPipelineStatus,
  body: (record CompilationPipelineStatus {totalTiersCount := 3, deadBranchesPruned := 14, simdVectorWidthBits := 512, zeroCostRustEmitted := true})
}

definition{
  name: isThreeTierPipeline,
  parameters: [status : CompilationPipelineStatus],
  result: Bool,
  body: (((project status totalTiersCount) =? 3) && (project status zeroCostRustEmitted))
}

definition{
  name: isSimdWidthValid,
  parameters: [status : CompilationPipelineStatus],
  result: Bool,
  body: (((project status simdVectorWidthBits) =? 256) || ((project status simdVectorWidthBits) =? 512))
}

theorem{
  name: pipeline_three_tier_sound,
  parameters: [],
  statement: ((isThreeTierPipeline (defaultPipelineStatus)) = true),
  proof: reflexivity{}
}

theorem{
  name: simd_vector_width_valid,
  parameters: [],
  statement: ((isSimdWidthValid (defaultPipelineStatus)) = true),
  proof: reflexivity{}
}

theorem{
  name: dead_branch_pruning_sound,
  parameters: [],
  statement: ((0 <=? (project (defaultPipelineStatus) deadBranchesPruned)) = true),
  proof: reflexivity{}
}
\end{semanticmodule}
